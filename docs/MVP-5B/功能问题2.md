### 修复 2：`RoleEditor.save` 未检查 HTTP 状态 + 无本地校验

**现状**：

```tsx
const r = await fetch('/api/weave/roles', { ... })
const d = (await r.json()) as { ok?: boolean; error?: string }
if (!d.ok) {
  setError(d.error ?? '保存失败')
  return
}
```

**问题**：
- `d.ok` 可能 undefined（后端 500 时返回 `{ error: '...' }` 无 `ok` 字段）→ 静默通过 → 用户以为保存成功
- 没有本地校验 → provider/model 空就提交

**修复**（`RoleEditor.tsx` 的 `save`）：

```tsx
// ★ 本地校验（提交前拦）
const validateLocal = (): string | null => {
  if (!form.id.trim()) return '角色 ID 不能为空'
  if (!form.name.trim()) return '名称不能为空'
  if (!form.provider.trim()) return '请选择 Provider'
  if (!form.model.trim()) return '请选择模型'
  return null
}

const save = async (): Promise<void> => {
  const localErr = validateLocal()
  if (localErr) { setError(localErr); return }

  setSaving(true); setError('')
  try {
    const r = await fetch('/api/weave/roles', { ... })
    // ★ 双检查：HTTP 状态 + 业务状态
    const d = (await r.json()) as { ok?: boolean; error?: string }
    if (!r.ok || d.ok === false) {
      setError(d.error ?? `保存失败（HTTP ${r.status}）`)
      return
    }
    onSaved?.()
    onClose()
  } catch (e) {
    setError('网络错误：' + (e instanceof Error ? e.message : String(e)))
  } finally {
    setSaving(false)
  }
}
```

### 修复 3：`CanvasEditor` 的 `initialGraph` 变更不生效（问题 7 根因）

**现状**：

```tsx
export function CanvasEditor({ initialGraph, ... }: Props) {
  const [nodes, setNodes] = useState<EditorNode[]>(() =>
    (initialGraph?.nodes ?? []).map(...)   // ★ 只在挂载时求值一次
  )
```

**问题**：`CanvasPane` 异步加载 graph 后 `setDraft(graph)` → 传新 `initialGraph` → **`useState` 初始值不再求值** → 画布保持空。

**这解释了两件事**：
- 第一次打开面板：图不显示（因为 draft 是异步来的）
- 刷新页面：重挂载后，`draft=null` 先初始化 → 空

**修复**：加 `useEffect` 同步。

```tsx
// src/client/dashboard/CanvasEditor.tsx
export function CanvasEditor({ initialGraph, onGraphChange, readonly = false, liveState, liveActivity }: Props) {
  const [nodes, setNodes] = useState<EditorNode[]>(() =>
    (initialGraph?.nodes ?? []).map((n, i) => ({ ... }))
  )
  const [edges, setEdges] = useState<EditorEdge[]>(() =>
    (initialGraph?.edges ?? []).filter(...).map(...)
  )

  // ★ 新增：initialGraph 变化时同步（首次异步加载后填充）
  const initialKeyRef = useRef<string>('')
  useEffect(() => {
    if (!initialGraph) return
    const key = `${initialGraph.entryPoint}|${initialGraph.nodes.length}|${initialGraph.edges.length}`
    if (initialKeyRef.current === key) return   // 同一个图，不覆盖（避免用户编辑被冲）
    initialKeyRef.current = key
    setNodes((initialGraph.nodes ?? []).map((n, i) => ({
      id: n.id,
      roleRef: n.roleRef ?? n.id,
      roleName: n.roleRef ?? n.id,
      x: (i % 3) * (NODE_W + COL_X),
      y: Math.floor(i / 3) * (NODE_H + ROW_Y),
      ...(n.artifactName !== undefined ? { artifactName: n.artifactName } : {}),
      ...(n.override !== undefined ? { override: n.override } : {}),
    })))
    setEdges((initialGraph.edges ?? []).filter((e) => e.type === 'seq' || e.type === 'cond' || e.type === 'loop').map((e, i) => ({
      id: `e-${i}-${e.from}-${e.to}`,
      from: e.from, to: e.to,
      type: (e.type === 'cond' || e.type === 'loop' ? e.type : 'seq') as EditorEdge['type'],
      ...(e.when !== undefined ? { when: e.when } : {}),
      ...(e.maxIter !== undefined ? { maxIter: e.maxIter } : {}),
    })))
  }, [initialGraph])

  // ...
}
```

**关键**：`initialKeyRef` 防止用户编辑后重渲染覆盖。只在"图和之前不一样"时同步。

### 修复 4：`routes.ts` start 检查 `result.ok`（问题 5 根因）

**现状**（`src/l4-visual/host/routes.ts` 第 ~370 行）：

```ts
const result = await runGraphRealTool(...)
updateTask(taskId, { graphId: result.graphId, status: 'running' })   // ← 不检查 result.ok
json(res, 200, { ok: true, taskId, graphId: result.graphId, status: 'running' })
```

**问题**：`runGraphRealTool` 有 `return { ok: false, message }` 分支（图校验失败等），**不抛错** → 被吞。

**修复**：

```ts
if (method === 'POST' && action === 'start') {
  if (!STARTABLE_STATUSES.has(task.status)) { json(res, 409, { error: `状态不允许: ${task.status}` }); return }
  if (!task.parentAgent) { json(res, 409, { error: '缺少 parent agent' }); return }

  const { runGraphRealTool } = await import('../../cli/graph-run-commands.js')
  try {
    const result = await runGraphRealTool(
      ctx, task.graph, task.userInput,
      task.parentAgent as never, task.outputDir,
    )
    // ★ 关键修复
    if (!result.ok) {
      updateTask(taskId, { status: 'failed', error: result.message })
      json(res, 400, { ok: false, error: result.message })
      return
    }
    if (!result.graphId) {
      updateTask(taskId, { status: 'failed', error: '内部错误：graphId 为空' })
      json(res, 500, { ok: false, error: '内部错误：graphId 为空' })
      return
    }
    updateTask(taskId, { graphId: result.graphId, status: 'running', startedAt: Date.now() })
    json(res, 200, { ok: true, taskId, graphId: result.graphId, status: 'running' })
  } catch (error) {
    const message = error instanceof Error ? error.message : String(error)
    updateTask(taskId, { status: 'failed', error: message })
    json(res, 500, { ok: false, error: message })
  }
  return
}
```

**顺带修前端**（`BoardToolbar.tsx`）：

## 四、P2 修复

### 修复 9：暂停/终止无法中断运行中子代理

**现状**：`graph-control` 改状态，但 `addSubagent` 里的 `waitForSubagentEnd` 只响应 `options.signal`（RunOptions），**不响应 graph-control**。

**修复 9a**（`src/l2-engine/graph-control.ts`）：

```ts
type ControlAction = 'pause' | 'resume' | 'stop'
const listeners = new Map<string, Set<(action: ControlAction) => void>>()

export function subscribeControl(graphId: string, cb: (action: ControlAction) => void): () => void {
  const set = listeners.get(graphId) ?? new Set()
  set.add(cb)
  listeners.set(graphId, set)
  return () => { set.delete(cb); if (set.size === 0) listeners.delete(graphId) }
}

export function controlActiveGraph(action: ControlAction, graphId: string): boolean {
  const ctrl = getGraphControl(graphId)
  if (action === 'pause') ctrl.pause()
  else if (action === 'resume') ctrl.resume()
  else ctrl.stop()
  // ★ 通知订阅者
  const set = listeners.get(graphId)
  if (set) for (const cb of set) cb(action)
  return true
}
```

**修复 9b**（`src/l2-engine/state-graph.ts` 的 `addSubagent`）：

```ts
// 在 waitForSubagentEnd 调用前
const abortCtl = new AbortController()
const unsubControl = subscribeControl(nodeCtx.graphId, (action) => {
  if (action === 'pause' || action === 'stop') abortCtl.abort()
})

try {
  result = await waitForSubagentEnd(nodeCtx.ctx, activeChildId, {
    ...(signal !== undefined ? { signal } : {}),
    // ★ 合并外部 signal 与控制 signal（用一个转发）
    ...(abortCtl.signal ? { signal: mergeAbort(signal, abortCtl.signal) } : {}),
    parentAgent: agent,
    // ... 其他回调
  })
} finally {
  unsubControl()
}

// 工具函数
function mergeAbort(a: AbortSignal | undefined, b: AbortSignal): AbortSignal {
  if (!a) return b
  const ctl = new AbortController()
  const onAbort = (): void => ctl.abort()
  if (a.aborted || b.aborted) ctl.abort()
  else { a.addEventListener('abort', onAbort, { once: true }); b.addEventListener('abort', onAbort, { once: true }) }
  return ctl.signal
}
```

## 八、一句话

**问题 1/3/7 是前端优化引入的回归**：

1. **RoleEditor 编辑已有角色没加载 provider/model** → 保存必失败
2. **CanvasEditor 的 initialGraph 不响应变化** → 异步加载的图不显示
3. **CanvasPane 服务器 graph 优先** → 用户编辑被初始模板覆盖

**问题 5 是原有的**：`routes.ts` 不检查 `result.ok`。

**修 4 处（0.9d），图能跑、Tab 不丢图、角色能存**。
